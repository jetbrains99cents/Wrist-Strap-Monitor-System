function Get-FolderTree {
    param(
        [string]$Path = (Get-Location),
        [int]$IndentLevel = 0,
        # Add or remove patterns here as needed!
        [string[]]$Exclude = @(".git", ".vscode", "__pycache__", ".venv", ".DS_Store", "*.log", "*.tmp", ".pytest_cache", ".idea", "*.uv.lock", "*.http", ".env", "*.pem")
    )

    $indent = "    " * $IndentLevel # 4 spaces per indent level

    Get-ChildItem -LiteralPath $Path -Force | ForEach-Object {
        $currentPath = $_.FullName
        $name = $_.Name

        # Skip excluded items
        if ($Exclude | Where-Object { $name -like $_ -or $currentPath -like $_ }) {
            return # Skip this item and continue
        }

        if ($_.PSIsContainer) { # Is a directory
            Write-Host "$($indent)├───$name" -ForegroundColor Cyan
            # Recursively call for sub-directories
            Get-FolderTree -Path $currentPath -IndentLevel ($IndentLevel + 1) -Exclude $Exclude
        } else { # Is a file
            Write-Host "$($indent)│   $name" -ForegroundColor Green
        }
    }
}
